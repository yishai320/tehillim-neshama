סֵדֶר הַקְּרִיאָה

פרק קי"ט בתהילים בנוי לפי סדר האלף-בית, ובכל אות שמונה פסוקים. נהגו לקרוא לעילוי נשמת המנוח את הפסוקים של האותיות המרכיבות את שמו, ואחריהם את אותיות המילה נְשָׁמָה. לאחר הקריאה אומרים "לאחר הקריאה", "יהי רצון" ותפילת "אל מלא רחמים".

לְאַחַר הַקְּרִיאָה

יְהִי רָצוֹן מִלְּפָנֶיךָ

יְהִי רָצוֹן מִלְּפָנֶיךָ יְהֹוָה אֱלֹהֵינוּ וֵאלֹהֵי אָבוֹתֵינוּ, שֶׁבִּזְכוּת מִזְמוֹרֵי הַתְּהִלִּים שֶׁקָּרָ֫אנוּ לְפָנֶיךָ, וּבִזְכוּת פְּסוּקֵיהֶם וְתֵיבוֹתֵיהֶם וְאוֹתִיּוֹתֵיהֶם וּנְקֻדּוֹתֵיהֶם, וְטַעֲמֵיהֶם, וּבִזְכוּת שְׁמוֹתֶיךָ הַקְּדוֹשִׁים וְהַטְּהוֹרִים הַיּוֹצְאִים מֵהֶם, שֶׁתְּכַפֶּר לָנוּ עַל כָּל חַטֹּאותֵינוּ וְתִסְלַח לָנוּ עַל כָּל עֲוֺנוֹתֵינוּ, וְתִמְחַל לָנוּ עַל כָּל פְּשָׁעֵינוּ שֶׁחָטָ֫אנוּ וְשֶׁעָוִ֫ינוּ וְשֶׁפָּשַׁ֫עְנוּ לְפָנֶיךָ. וְהַחֲזִירֵנוּ בִּתְשׁוּבָה שְׁלֵמָה לְפָנֶיךָ, וְהַדְרִיכֵנוּ לַעֲבוֹדָתֶךָ וְתִפְתַּח לִבֵּנוּ בְּתַלְמוּד תּוֹרָתֶךָ. וְתִשְׁלַח רְפוּאָה שְׁלֵמָה לְחוֹלֵי עַמֶּךָ. וְתִקְרָא לַשְּׁבוּיִים דְּרוֹר וְלַאֲסוּרִים פְּקַח קוֹחַ, וּלְכָל הוֹלְכֵי דְרָכִים וְעוֹבְרֵי יַמִּים וּנְהָרוֹת מֵעַמְּךָ יִשְׂרָאֵל תַּצִּילֵם מִכָּל צַעַר וְנֶזֶק, וְתַגִּיעֵם לִמְחוֹז חֶפְצָם לְחַיִּים וּלְשָׁלוֹם. וְתִפְקֹד לְכָל חֲשׂוּכֵי בָנִים בְּזֶרַע שֶׁל קַיָּמָא לַעֲבוֹדָתֶךָ וּלְיִרְאָתֶךָ. וּתְבַטֵּל מֵעָלֵינוּ וּמִכָּל עַמְּךָ בֵּית יִשְׂרָאֵל, בְּכָל מָקוֹם שֶׁהֵם, כָּל גְּזֵרוֹת קָשׁוֹת וְרָעוֹת, וְתַטֵּה לֵב הַמַּלְכוּת עָלֵינוּ לְטוֹבָה. וְתִשְׁלַח בְּרָכָה וְהַצְלָחָה בְּכָל מַעֲשֵׂה יָדֵינוּ, וְהָכֵן פַּרְנָסָתֵנוּ מִיָּדְךָ הָרְחָבָה וְהַמְּלֵאָה, וְלֹא יִצְטָרְכוּ עַמְּךָ בֵּית יִשְׂרָאֵל זֶה לָזֶה וְלֹא לְעַם אַחֵר. וּתְמַהֵר וְתָחִישׁ לִגְאֻלָּ֫תֵנוּ וְתִבְנֶה בֵּית מִקְדָּשֵׁנוּ וְתִפְאַרְתֵּנוּ. וּבִזְכוּת שְׁלוֹשׁ עֶשְׂרֵה מִדּוֹתֶיךָ שֶׁל רַחֲמִים הַכְּתוּבִים בְּתוֹרָתֶךָ, שֶׁנֶּאֱמַר: יְהֹוָה, יְהֹוָה, אֵל רַחוּם וְחַנּוּן, אֶרֶךְ אַפַּיִם וְרַב חֶסֶד וֶאֱמֶת, נֹצֵר חֶסֶד לָאֲלָפִים נֹשֵׂא עָוֺן וָפֶשַׁע וְחַטָּאָה וְנַקֵּה׃ עָזְרֵנוּ אֱלֹהֵי יִשְׁעֵנוּ עַל דְּבַר כְּבוֹד שְׁמֶךָ, וְהַצִּילֵנוּ וְכַפֵּר עַל חַטֹּאותֵינוּ לְמַעַן שְׁמֶךָ׃ בָּרוּךְ יְהֹוָה לְעוֹלָם, אָמֵן וְאָמֵן׃

אֵל מָלֵא רַחֲמִים

התאמה אישית של תפילת האזכרה:
(הטקסט יתעדכן אוטומטית למטה)

אֵל מָלֵא רַחֲמִים שׁוֹכֵן בַּמְּרוֹמִים, הַמְצֵא מְנוּחָה נְכוֹנָה עַל כַּנְפֵי הַשְּׁכִינָה, בְּמַעֲלוֹת קְדוֹשִׁים וּטְהוֹרִים כְּזוֹהַר הָרָקִיעַ מַזְהִירִים, לְנִשְׁמַת יְרַחְמִיאֵל דָּוִד בֶּן __________ שֶׁהָלַךְ לְעוֹלָמוֹ, עֲבוּר שֶׁנָּדְבוּ צְדָקָה בְּעַד הַזְכָּרַת נִשְׁמָתוֹ, בְּגַן עֵדֶן תְּהֵא מְנוּחָתוֹ. לָכֵן בַּעַל הָרַחֲמִים יַסְתִּירֵהוּ בְּסֵתֶר כְּנָפָיו לְעוֹלָמִים, וְיִצְרֹר בִּצְרוֹר הַחַיִּים אֶת נִשְׁמָתוֹ, יְהֹוָה הוּא נַחֲלָתוֹ, וְיָנוּחַ בְּשָׁלוֹם עַל מִשְׁכָּבוֹ, וְנֹאמַר אָמֵן.
